Chapter 1

Introduction to DSA

Data Structures & Algorithms (DSA) is the heart of efficient software. A data structure organizes data so it can be accessed and modified efficiently. An algorithm is a step-by-step procedure for solving a problem. Choosing the right combination can turn a program that takes hours into one that takes milliseconds.

Why DSA matters

  • Performance: The right algorithm handles millions of records; the wrong one chokes on thousands.
  • Interviews: DSA is the standard lens for evaluating engineers at top tech companies.
  • Foundation: Databases, compilers, operating systems, and AI all rely on classic data structures and algorithms.

How to use this tutorial

Every code block is runnable Python. Copy it, paste it into a file (e.g. dsa_demo.py), and run it with python dsa_demo.py. Each example prints clear, meaningful output so you can see the behavior with your own eyes.

Complexity Analysis (Big O)

Big O notation describes how runtime (or memory) grows as input size n grows. It lets you compare algorithms independent of machine speed.

Common complexity classes

Big O Name Growth Example
O(1)ConstantFlatArray index access
O(log n)LogarithmicSlow growthBinary search
O(n)LinearProportional to nSingle loop
O(n log n)LinearithmicSlightly worse than linearMerge sort
O(n²)QuadraticGrows fastNested loops
O(2ⁿ)ExponentialExplodes quicklyNaive recursion (Fibonacci)

Timing O(1) vs O(n) vs O(n²)

The following demonstrates how work scales with input size.

"""Complexity demonstration: O(1), O(n), and O(n^2).
Run: python complexity_demo.py
"""

import time

def constant_time(arr):
    """O(1): Always one operation, regardless of array size."""
    return arr[0] if arr else None

def linear_time(arr, target):
    """O(n): Check each element once."""
    for item in arr:
        if item == target:
            return True
    return False

def quadratic_time(arr):
    """O(n^2): Nested loop over all pairs."""
    count = 0
    for i in arr:
        for j in arr:
            count += 1
    return count

def time_it(fn, *args):
    start = time.perf_counter()
    result = fn(*args)
    elapsed = (time.perf_counter() - start) * 1000  # milliseconds
    return result, elapsed

# Build arrays of different sizes
sizes = [1_000, 5_000, 10_000]
print(f"{'Size':>8} | {'O(1) ms':>9} | {'O(n) ms':>9} | {'O(n^2) ms':>11}")
print("-" * 46)

for n in sizes:
    arr = list(range(n))

    _, t1 = time_it(constant_time, arr)
    _, t2 = time_it(linear_time, arr, n - 1)  # worst case: last element
    _, t3 = time_it(quadratic_time, arr)

    print(f"{n:>8} | {t1:>9.4f} | {t2:>9.4f} | {t3:>11.4f}")

print("\nNotice: O(n^2) grows much faster than O(n) or O(1).")
# Sample output:
# Size    |   O(1) ms |   O(n) ms |  O(n^2) ms
# ----------------------------------------------
#    1000 |    0.0012 |    0.0234 |     3.4125
#    5000 |    0.0008 |    0.0412 |    85.2103
#   10000 |    0.0005 |    0.0987 |   342.1055

Arrays & Lists

A list (Python's dynamic array) stores elements contiguously. Access by index is O(1); appending is amortized O(1); inserting/deleting in the middle is O(n) because elements must shift.

List operations demo

"""Arrays / lists in Python.
Run: python arrays_demo.py
"""

# Create a list
nums = [10, 20, 30, 40, 50]
print("Original list:", nums)

# O(1) access by index
print("Index 2:", nums[2])  # 30

# Append (amortized O(1))
nums.append(60)
print("After append 60:", nums)

# Insert at index (O(n) - elements shift)
nums.insert(1, 15)
print("After insert 15 at index 1:", nums)

# Remove by value (O(n) - search + shift)
nums.remove(30)
print("After remove 30:", nums)

# Pop from end (O(1))
last = nums.pop()
print("Popped last:", last, "| list now:", nums)

# Slicing creates a new list (O(k))
sub = nums[1:4]
print("Slice [1:4]:", sub)

# List comprehension (idiomatic Python)
squares = [x * x for x in range(1, 6)]
print("Squares 1..5:", squares)

# Popped last: 60 | list now: [10, 15, 20, 40, 50]
# Slice [1:4]: [15, 20, 40]
# Squares 1..5: [1, 4, 9, 16, 25]

Two-sum problem (O(n) with a hash set)

Given a list and a target, find two indices whose values sum to the target.

"""Two-sum: find two numbers that add to target. O(n) time, O(n) space.
Run: python two_sum.py
"""

def two_sum(nums, target):
    """Return indices of two numbers that sum to target, or None."""
    seen = {}  # value -> index
    for i, num in enumerate(nums):
        complement = target - num
        if complement in seen:          # O(1) lookup
            return [seen[complement], i]
        seen[num] = i                   # store for later lookups
    return None

# Demo
test_cases = [
    ([2, 7, 11, 15], 9),   # -> [0, 1] (2 + 7)
    ([3, 2, 4], 6),        # -> [1, 2] (2 + 4)
    ([3, 3], 6),           # -> [0, 1] (3 + 3)
    ([1, 2, 3], 7),        # -> None
]

for nums, target in test_cases:
    result = two_sum(nums, target)
    print(f"nums={nums}, target={target} -> indices={result}")
# Output:
# nums=[2, 7, 11, 15], target=9 -> indices=[0, 1]
# nums=[3, 2, 4], target=6 -> indices=[1, 2]
# nums=[3, 3], target=6 -> indices=[0, 1]
# nums=[1, 2, 3], target=7 -> indices=None

Chapter 4

Linked Lists

A linked list is a chain of nodes. Each node holds data and a reference to the next node. Insertion/deletion at the head is O(1), but access by index is O(n) because you must walk from the head.

Singly linked list implementation

"""Singly linked list with append, prepend, delete, and print.
Run: python linked_list.py
"""

class Node:
    """A single node: holds data and a link to the next node."""
    def __init__(self, data):
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        self.head = None

    def prepend(self, data):
        """Insert at the front — O(1)."""
        new_node = Node(data)
        new_node.next = self.head
        self.head = new_node

    def append(self, data):
        """Insert at the end — O(n) because we walk to the tail."""
        new_node = Node(data)
        if not self.head:
            self.head = new_node
            return
        current = self.head
        while current.next:
            current = current.next
        current.next = new_node

    def delete(self, key):
        """Delete first node with matching data — O(n)."""
        current = self.head
        # Case: head holds the key
        if current and current.data == key:
            self.head = current.next
            current = None
            return
        # Search for the key, keeping track of previous node
        prev = None
        while current and current.data != key:
            prev = current
            current = current.next
        if not current:  # key not found
            return
        prev.next = current.next
        current = None

    def __str__(self):
        parts = []
        current = self.head
        while current:
            parts.append(str(current.data))
            current = current.next
        return " -> ".join(parts) if parts else "empty"

# Demo
ll = LinkedList()
ll.append(10)
ll.append(20)
ll.append(30)
print("After appends 10,20,30:", ll)        # 10 -> 20 -> 30

ll.prepend(5)
print("After prepend 5:    ", ll)           # 5 -> 10 -> 20 -> 30

ll.delete(20)
print("After delete 20:    ", ll)           # 5 -> 10 -> 30

ll.delete(99)  # not present
print("After delete 99:    ", ll)           # unchanged
# Output:
# After appends 10,20,30: 10 -> 20 -> 30
# After prepend 5:     5 -> 10 -> 20 -> 30
# After delete 20:     5 -> 10 -> 30
# After delete 99:     5 -> 10 -> 30

Visualization — Singly linked node chain

Chapter 5

Stacks & Queues

Stack: LIFO (last-in, first-out). Push/pop from the top. Used for undo, call stack, DFS.
Queue: FIFO (first-in, first-out). Enqueue at back, dequeue at front. Used for BFS, task scheduling.

Stack demo (list-based)

"""Stack: Last-In First-Out (LIFO).
Run: python stack_demo.py
"""

class Stack:
    def __init__(self):
        self.items = []

    def push(self, item):
        self.items.append(item)          # O(1)

    def pop(self):
        if self.is_empty():
            raise IndexError("pop from empty stack")
        return self.items.pop()          # O(1)

    def peek(self):
        if self.is_empty():
            return None
        return self.items[-1]            # O(1)

    def is_empty(self):
        return len(self.items) == 0

    def __str__(self):
        return " -> ".join(str(x) for x in reversed(self.items)) or "empty"

# Undo-style demo
stack = Stack()
for action in ["type 'Hello'", "type ' World'", "bold", "italic"]:
    stack.push(action)
    print(f"Pushed: {action}")

print("\nStack top to bottom:", stack)
print("Undo (pop):", stack.pop())   # italic
print("Undo (pop):", stack.pop())   # bold
print("Stack now:", stack)
# Output:
# Pushed: type 'Hello'
# Pushed: type ' World'
# Pushed: bold
# Pushed: italic
#
# Stack top to bottom: italic -> bold -> type ' World' -> type 'Hello'
# Undo (pop): italic
# Undo (pop): bold
# Stack now: type ' World' -> type 'Hello'

Queue demo (collections.deque)

"""Queue: First-In First-Out (FIFO) using deque for O(1) popleft.
Run: python queue_demo.py
"""

from collections import deque

class Queue:
    def __init__(self):
        self.items = deque()

    def enqueue(self, item):
        self.items.append(item)          # O(1)

    def dequeue(self):
        if self.is_empty():
            raise IndexError("dequeue from empty queue")
        return self.items.popleft()      # O(1)

    def peek(self):
        return self.items[0] if not self.is_empty() else None

    def is_empty(self):
        return len(self.items) == 0

    def __str__(self):
        return " <- ".join(str(x) for x in self.items) or "empty"

# Print-job demo
q = Queue()
for job in ["doc1.pdf", "doc2.pdf", "doc3.pdf"]:
    q.enqueue(job)
    print(f"Enqueued: {job}")

print("\nQueue front to back:", q)
print("Printing (dequeue):", q.dequeue())  # doc1.pdf
print("Printing (dequeue):", q.dequeue())  # doc2.pdf
print("Queue now:", q)
# Output:
# Enqueued: doc1.pdf
# Enqueued: doc2.pdf
# Enqueued: doc3.pdf
#
# Queue front to back: doc1.pdf <- doc2.pdf <- doc3.pdf
# Printing (dequeue): doc1.pdf
# Printing (dequeue): doc2.pdf
# Queue now: doc3.pdf

Hash Tables

A hash table maps keys to values using a hash function. Python's dict and set are hash tables. Average-case O(1) insert, lookup, and delete. Collisions occur when two keys hash to the same slot; good hash functions minimize them.

Hash table demo (Python dict)

Run: python hash_demo.py
"""

    "Charlie": "555-9012",

"""Count word frequencies using a hash map. O(n) time.

def word_frequencies(text):

Chapter 7

Trees & Binary Search Trees

A tree is a hierarchy of nodes with a root and children. A binary search tree (BST) keeps, for every node, all left descendants smaller and all right descendants larger. This makes search, insert, and delete O(h) where h is tree height (O(log n) if balanced).

Binary search tree implementation

"""Binary Search Tree (BST): insert, search, in-order traversal.
Run: python bst_demo.py
"""

class TreeNode:
    def __init__(self, key):
        self.key = key
        self.left = None
        self.right = None

class BST:
    def __init__(self):
        self.root = None

    def insert(self, key):
        """Insert a key — O(h) where h is tree height."""
        self.root = self._insert(self.root, key)

    def _insert(self, node, key):
        if node is None:
            return TreeNode(key)
        if key < node.key:
            node.left = self._insert(node.left, key)
        elif key > node.key:
            node.right = self._insert(node.right, key)
        return node

    def search(self, key):
        """Search for a key — O(h)."""
        return self._search(self.root, key)

    def _search(self, node, key):
        if node is None:
            return False
        if key == node.key:
            return True
        if key < node.key:
            return self._search(node.left, key)
        return self._search(node.right, key)

    def inorder(self):
        """Return keys in sorted order (left, root, right)."""
        result = []
        self._inorder(self.root, result)
        return result

    def _inorder(self, node, result):
        if node:
            self._inorder(node.left, result)
            result.append(node.key)
            self._inorder(node.right, result)

# Demo: insert in unsorted order, then traverse sorted
bst = BST()
for key in [50, 30, 70, 20, 40, 60, 80]:
    bst.insert(key)

print("In-order traversal (sorted):", bst.inorder())
# [20, 30, 40, 50, 60, 70, 80]

print("Search 40:", bst.search(40))  # True
print("Search 99:", bst.search(99))  # False
# Output:
# In-order traversal (sorted): [20, 30, 40, 50, 60, 70, 80]
# Search 40: True
# Search 99: False

Visualization — Binary search tree

Search/insert/delete: O(h). Balanced tree: h = O(log n).

Chapter 8

Heaps & Priority Queues

A heap is a complete binary tree where each parent is ordered relative to its children (min-heap: parent ≤ children). Heap operations (push/pop) are O(log n). Ideal for priority queues and Dijkstra's algorithm. Python's heapq module is a min-heap.

Min-heap demo (heapq)

"""Min-heap / priority queue with heapq.
Run: python heap_demo.py
"""

import heapq

class MinHeap:
    def __init__(self):
        self.heap = []

    def push(self, item):
        heapq.heappush(self.heap, item)   # O(log n)

    def pop(self):
        return heapq.heappop(self.heap)   # O(log n)

    def peek(self):
        return self.heap[0] if self.heap else None  # O(1)

    def __str__(self):
        return str(self.heap)

# Demo: task scheduler with priorities (lower number = higher priority)
tasks = MinHeap()
tasks.push((3, "Low priority email"))
tasks.push((1, "Critical server alert"))
tasks.push((2, "Code review"))
tasks.push((1, "Security patch"))

print("Heap contents:", tasks)
print("Next task (pop):", tasks.pop())   # (1, 'Critical server alert')
print("Next task (pop):", tasks.pop())   # (1, 'Security patch')
print("Next task (pop):", tasks.pop())   # (2, 'Code review')
print("Next task (pop):", tasks.pop())   # (3, 'Low priority email')
# Output:
# Heap contents: [(1, 'Critical server alert'), (1, 'Security patch'), (2, 'Code review'), (3, 'Low priority email')]
# Next task (pop): (1, 'Critical server alert')
# Next task (pop): (1, 'Security patch')
# Next task (pop): (2, 'Code review')
# Next task (pop): (3, 'Low priority email')

Chapter 9

Graphs & Traversal

A graph is a set of nodes (vertices) connected by edges. Two fundamental traversals:
• BFS (Breadth-First Search): explore layer by layer using a queue — finds shortest paths in unweighted graphs.
• DFS (Depth-First Search): dive deep using recursion or a stack — useful for cycles, connectivity, topological order.

BFS & DFS on an adjacency list

"""Graph traversals: BFS and DFS.
Run: python graph_demo.py
"""

from collections import deque

# Graph as adjacency list
graph = {
    "A": ["B", "C"],
    "B": ["D", "E"],
    "C": ["F"],
    "D": [],
    "E": ["F"],
    "F": [],
}

def bfs(start):
    """Breadth-First Search using a queue."""
    visited = set()
    queue = deque([start])
    order = []
    while queue:
        node = queue.popleft()
        if node not in visited:
            visited.add(node)
            order.append(node)
            queue.extend(graph[node])
    return order

def dfs(start, visited=None, order=None):
    """Depth-First Search using recursion."""
    if visited is None:
        visited = set()
    if order is None:
        order = []
    visited.add(start)
    order.append(start)
    for neighbor in graph[start]:
        if neighbor not in visited:
            dfs(neighbor, visited, order)
    return order

print("BFS from A:", bfs("A"))    # A, B, C, D, E, F
print("DFS from A:", dfs("A"))    # A, B, D, E, F, C
# Output:
# BFS from A: ['A', 'B', 'C', 'D', 'E', 'F']
# DFS from A: ['A', 'B', 'D', 'E', 'F', 'C']

Chapter 10

Sorting Algorithms

Sorting is a building block for search, data analysis, and interview problems. Below are the classic algorithms with their complexities.

Bubble sort, insertion sort, merge sort, quick sort

"""Sorting algorithms demo.
Run: python sorting_demo.py
"""

def bubble_sort(arr):
    """O(n^2). Repeatedly swaps adjacent out-of-order elements."""
    a = arr[:]
    n = len(a)
    for i in range(n):
        for j in range(0, n - i - 1):
            if a[j] > a[j + 1]:
                a[j], a[j + 1] = a[j + 1], a[j]
    return a

def insertion_sort(arr):
    """O(n^2). Builds sorted array one element at a time."""
    a = arr[:]
    for i in range(1, len(a)):
        key = a[i]
        j = i - 1
        while j >= 0 and a[j] > key:
            a[j + 1] = a[j]
            j -= 1
        a[j + 1] = key
    return a

def merge_sort(arr):
    """O(n log n). Divide, recursively sort, merge."""
    if len(arr) <= 1:
        return arr
    mid = len(arr) // 2
    left = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])
    return _merge(left, right)

def _merge(left, right):
    result = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i]); i += 1
        else:
            result.append(right[j]); j += 1
    result.extend(left[i:])
    result.extend(right[j:])
    return result

def quick_sort(arr):
    """O(n log n) average, O(n^2) worst. Pivot-based partitioning."""
    if len(arr) <= 1:
        return arr
    pivot = arr[len(arr) // 2]
    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]
    return quick_sort(left) + middle + quick_sort(right)

data = [64, 34, 25, 12, 22, 11, 90]
print("Original:", data)
print("Bubble sort:", bubble_sort(data))
print("Insertion sort:", insertion_sort(data))
print("Merge sort:", merge_sort(data))
print("Quick sort:", quick_sort(data))
# Output:
# Original: [64, 34, 25, 12, 22, 11, 90]
# Bubble sort: [11, 12, 22, 25, 34, 64, 90]
# Insertion sort: [11, 12, 22, 25, 34, 64, 90]
# Merge sort: [11, 12, 22, 25, 34, 64, 90]
# Quick sort: [11, 12, 22, 25, 34, 64, 90]

Visualization — Merge sort divide & conquer

Chapter 11

Recursion & Backtracking

Recursion: a function calling itself with a smaller input, until a base case stops it. Backtracking is recursion that explores choices and undoes (backs out of) dead ends — used for permutations, N-queens, sudoku.

Factorial, Fibonacci, and permutations

"""Recursion and backtracking demos.
Run: python recursion_demo.py
"""

def factorial(n):
    """n! = n * (n-1)! ; base case 0! = 1"""
    if n == 0:
        return 1
    return n * factorial(n - 1)

def fibonacci(n):
    """Naive recursive Fibonacci — O(2^n). Use DP for speed."""
    if n <= 1:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

def permutations(s):
    """All rearrangements of a string via backtracking."""
    if len(s) <= 1:
        return [s]
    result = []
    for i, ch in enumerate(s):
        rest = s[:i] + s[i+1:]
        for perm in permutations(rest):
            result.append(ch + perm)
    return result

print("Factorial of 5:", factorial(5))      # 120
print("Fibonacci of 8:", fibonacci(8))      # 21
print("Permutations of 'abc':", permutations("abc"))
# ['abc', 'acb', 'bac', 'bca', 'cab', 'cba']
# Output:
# Factorial of 5: 120
# Fibonacci of 8: 21
# Permutations of 'abc': ['abc', 'acb', 'bac', 'bca', 'cab', 'cba']

Chapter 12

Dynamic Programming

Dynamic programming (DP) solves overlapping subproblems once and stores results (memoization, top-down, or tabulation, bottom-up). It turns exponential recursion into polynomial time.

Fibonacci: naive vs memoized vs tabulated

"""Dynamic programming: Fibonacci three ways.
Run: python dp_demo.py
"""

# Naive recursion — O(2^n), slow
def fib_naive(n):
    if n <= 1:
        return n
    return fib_naive(n - 1) + fib_naive(n - 2)

# Memoization (top-down) — O(n)
from functools import lru_cache
@lru_cache(maxsize=None)
def fib_memo(n):
    if n <= 1:
        return n
    return fib_memo(n - 1) + fib_memo(n - 2)

# Tabulation (bottom-up) — O(n) time, O(n) space
def fib_tab(n):
    if n <= 1:
        return n
    dp = [0] * (n + 1)
    dp[1] = 1
    for i in range(2, n + 1):
        dp[i] = dp[i - 1] + dp[i - 2]
    return dp[n]

n = 40
print(f"fib_naive({n}) =", fib_naive(n))      # slow but correct
print(f"fib_memo({n})  =", fib_memo(n))       # instant
print(f"fib_tab({n})   =", fib_tab(n))        # instant
# All print: 102334155
# Output:
# fib_naive(40) = 102334155
# fib_memo(40)  = 102334155
# fib_tab(40)   = 102334155

Greedy Algorithms

A greedy algorithm makes the locally optimal choice at each step, hoping it leads to a global optimum. Works when the problem has the greedy-choice property (e.g. activity selection, coin change with canonical coins). Not always correct — that's when DP or other methods are needed.

Coin change (greedy, canonical coins)

"""Greedy coin change: fewest coins for canonical denominations.
Run: python greedy_demo.py
"""

def coin_change_greedy(amount, coins):
    """Return dict {coin: count} using largest coins first."""
    coins = sorted(coins, reverse=True)
    result = {}
    remaining = amount
    for coin in coins:
        count = remaining // coin
        if count:
            result[coin] = count
            remaining -= count * coin
    if remaining != 0:
        return None  # cannot make exact change with these coins
    return result

amount = 93
coins = [1, 5, 10, 25]  # US coins (canonical: greedy works)
change = coin_change_greedy(amount, coins)
print(f"Amount: {amount}")
print(f"Coins:  {coins}")
print(f"Change: {change}")
print(f"Total coins: {sum(change.values())}")
# Output:
# Amount: 93
# Coins:  [1, 5, 10, 25]
# Change: {25: 3, 10: 1, 5: 1, 1: 3}
# Total coins: 8  (3*25 + 1*10 + 1*5 + 3*1 = 93)

"""Route & Task Planner capstone.
Graph + priority queue + hash table. Run: python planner.py
"""

from collections import deque
import heapq

# --- Graph: cities and roads with distances ---
class CityGraph:
    def __init__(self):
        self.adj = {}   # city -> list of (neighbor, distance)

    def add_road(self, a, b, dist):
        self.adj.setdefault(a, []).append((b, dist))
        self.adj.setdefault(b, []).append((a, dist))

    def bfs_reachable(self, start):
        """Cities reachable from start (unweighted reachability)."""
        visited = {start}
        queue = deque([start])
        while queue:
            city = queue.popleft()
            for neighbor, _ in self.adj.get(city, []):
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)
        return sorted(visited)

    def dijkstra(self, start, goal):
        """Shortest path by distance using a min-heap."""
        dist = {start: 0}
        prev = {start: None}
        heap = [(0, start)]
        while heap:
            d, city = heapq.heappop(heap)
            if city == goal:
                break
            if d > dist.get(city, float('inf')):
                continue
            for neighbor, w in self.adj.get(city, []):
                nd = d + w
                if nd < dist.get(neighbor, float('inf')):
                    dist[neighbor] = nd
                    prev[neighbor] = city
                    heapq.heappush(heap, (nd, neighbor))
        # Reconstruct path
        path = []
        at = goal
        while at is not None:
            path.append(at)
            at = prev.get(at)
        path.reverse()
        return dist.get(goal, float('inf')), path if path and path[0] == start else []

# --- Task scheduler: min-heap by priority ---
class TaskScheduler:
    def __init__(self):
        self.heap = []
        self.counter = 0  # tie-breaker so equal priorities stay ordered

    def add(self, task, priority):
        heapq.heappush(self.heap, (priority, self.counter, task))
        self.counter += 1

    def next_task(self):
        if not self.heap:
            return None
        _, _, task = heapq.heappop(self.heap)
        return task

    def __str__(self):
        return ", ".join(t for _, _, t in sorted(self.heap)) or "none"

# --- Interactive planner ---
def main():
    graph = CityGraph()
    for a, b, d in [
        ("NYC", "Philly", 95), ("Philly", "DC", 140),
        ("DC", "Richmond", 110), ("NYC", "Boston", 215),
        ("Boston", "Providence", 50), ("Providence", "Hartford", 100),
        ("Hartford", "NYC", 120),
    ]:
        graph.add_road(a, b, d)

    scheduler = TaskScheduler()

    print("=== Route & Task Planner ===\n")
    while True:
        print("1. Add a task")
        print("2. Show next task")
        print("3. Find shortest route between cities")
        print("4. List reachable cities from a city")
        print("5. Exit")
        choice = input("\nChoose (1-5): ").strip()

        if choice == "1":
            task = input("Task description: ").strip()
            try:
                priority = int(input("Priority (1=highest, 5=lowest): "))
            except ValueError:
                priority = 3
            scheduler.add(task, priority)
            print(f"  ✅ Added: '{task}' (priority {priority})\n")
        elif choice == "2":
            task = scheduler.next_task()
            if task:
                print(f"  ▶ Next: {task}\n")
            else:
                print("  No tasks scheduled.\n")
        elif choice == "3":
            start = input("From city: ").strip().upper()
            goal = input("To city: ").strip().upper()
            dist, path = graph.dijkstra(start, goal)
            if path:
                print(f"  Shortest route {start} -> {goal}: {' -> '.join(path)} ({dist} miles)\n")
            else:
                print(f"  No route from {start} to {goal}.\n")
        elif choice == "4":
            city = input("City: ").strip().upper()
            reachable = graph.bfs_reachable(city)
            print(f"  From {city}, reachable: {', '.join(reachable)}\n")
        elif choice == "5":
            print("\nGoodbye! 🎓\n")
            break
        else:
            print("  Invalid choice.\n")

if __name__ == "__main__":
    main()
